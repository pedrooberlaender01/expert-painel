Create `src/hooks/useSectionGate.ts` following the pattern of `useFeatureGate.ts`:

```typescript
import { useAuthStore } from '../stores/authStore';
import { Navigate } from 'react-router-dom';
import React from 'react';

// Section keys matching secoes_habilitadas JSONB keys
export type SectionKey = 'dashboard' | 'conversas' | 'leads' | 'grupos' | 'envios' | 'torneios' | 'mensagens' | 'central_whatsapp';

// Maps route paths to section keys for gating
// Includes sub-routes that belong to a parent section
export const SECTION_PATH_MAP: Record<string, SectionKey> = {
  '/dashboard': 'dashboard',
  '/conversas': 'conversas',
  '/leads': 'leads',
  '/funil': 'leads',           // funil is part of Leads Assistente
  '/grupos': 'grupos',
  '/envios': 'envios',
  '/envios/simulador': 'envios',
  '/envios/historico': 'envios',
  '/envios/templates': 'envios',
  '/envios/agendamentos': 'envios',
  '/envios/agendados': 'envios',
  '/envios/gerar-copy': 'envios',
  '/torneios': 'torneios',
  '/mensagens': 'mensagens',
  '/central-whatsapp': 'central_whatsapp',
  // /notificacoes and /configuracoes are NOT mapped — always accessible per D-11
};

// Check if a specific section is enabled for the current expert
export function useSectionGate(section: SectionKey): boolean {
  const { user, impersonatedExpert } = useAuthStore();
  const expert = impersonatedExpert || user?.expert;

  // No expert = no gating (admin without impersonation sees everything)
  if (!expert) return true;

  // null secoes_habilitadas = all enabled per D-02
  if (!expert.secoes_habilitadas) return true;

  // Explicit check — default to true if key missing
  return expert.secoes_habilitadas[section] !== false;
}

// Check multiple sections at once (for Sidebar)
export function useSectionGates(sections: SectionKey[]): Record<SectionKey, boolean> {
  const { user, impersonatedExpert } = useAuthStore();
  const expert = impersonatedExpert || user?.expert;

  const result = {} as Record<SectionKey, boolean>;

  for (const section of sections) {
    if (!expert || !expert.secoes_habilitadas) {
      result[section] = true;
      continue;
    }
    result[section] = expert.secoes_habilitadas[section] !== false;
  }

  return result;
}

// Route guard component — wraps route elements, redirects to /dashboard if section disabled
// Per D-10: silent redirect, no error message
interface SectionGuardProps {
  section: SectionKey;
  children: React.ReactNode;
}

export function SectionGuard({ section, children }: SectionGuardProps) {
  const enabled = useSectionGate(section);
  if (!enabled) return React.createElement(Navigate, { to: '/dashboard', replace: true });
  return React.createElement(React.Fragment, null, children);
}
```

**Important notes:**
- Admin users (role=admin, no impersonation) should NOT be gated — they see everything. The `if (!expert) return true` handles this since admin without impersonation has `expert = null`.
- When admin is impersonating, they see what the expert sees (respects gating) — this is consistent with how impersonation works for plan limits.
