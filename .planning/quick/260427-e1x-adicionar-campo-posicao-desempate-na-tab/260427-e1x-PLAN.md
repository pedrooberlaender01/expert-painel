    2) Adicionar handler de swap (apos os outros handlers, antes do return):
    ```typescript
    const [swapLoading, setSwapLoading] = useState<string | null>(null);

    const handleSwapDesempate = async (currentId: string, otherId: string) => {
      if (!currentId || !otherId || swapLoading) return;
      const a = participantes.find(p => p.participante_id === currentId);
      const b = participantes.find(p => p.participante_id === otherId);
      if (!a || !b) return;
      setSwapLoading(currentId);
      try {
        // Atribui valores se algum estiver NULL — usa indices baseados no grupo
        const ap = a.posicao_desempate;
        const bp = b.posicao_desempate;
        let newA: number; let newB: number;
        if (ap != null && bp != null) { newA = bp; newB = ap; }
        else {
          // Atribui valores incrementais: o "other" (que vai para cima) recebe valor menor
          // currentId esta clicando ▲ ou ▼ — ja decidido pelo chamador qual vai pra cima
          // Convencao: chamador passa (current, other) onde current e o que clicou; vamos sempre dar
          // valor menor para o que ficara em cima na ordenacao final.
          // Estrategia: trocar significa current sobe (clicou ▲) -> current vira valor menor.
          // Mas tambem existe ▼ -> current vira valor maior. O chamador resolve isso passando args corretos.
          // Aqui apenas garantimos numeros: usar 1 e 2 quando ambos NULL, ou +/- 1 do existente.
          if (ap == null && bp == null) { newA = 1; newB = 2; }
          else if (ap == null) { newA = (bp as number) - 1; newB = bp as number; }
          else { newA = ap as number; newB = (ap as number) + 1; }
        }
        // Persistir em transacao logica (duas updates)
        const { error: e1 } = await supabase.from('participantes').update({ posicao_desempate: newA }).eq('id', currentId);
        if (e1) throw e1;
        const { error: e2 } = await supabase.from('participantes').update({ posicao_desempate: newB }).eq('id', otherId);
        if (e2) throw e2;
        showToast('success', 'Ordem de desempate atualizada');
        await fetchParticipantes();
      } catch (err: any) {
        showToast('error', err?.message || 'Erro ao trocar ordem');
      } finally {
        setSwapLoading(null);
      }
    };
    ```
